// Client Select Search Logic let selectedClient = null; function setupClientSearch(clientsData) { const searchInput = document.getElementById('clientSearchInput'); const dropdownList = document.getElementById('clientDropdownList'); if (!searchInput || !dropdownList) return; // Render list helper function renderList(items) { dropdownList.innerHTML = ''; if (items.length === 0) { dropdownList.innerHTML = `
لا توجد نتائج مطابقة
`; return; } items.forEach(client => { const item = document.createElement('div'); item.className = 'p-2.5 hover:bg-gray-100 dark:hover:bg-gray-700 cursor-pointer text-sm rounded-lg transition-colors flex justify-between items-center'; item.innerHTML = `${client.name} ${client.phone || ''}`; // Use mousedown instead of click to prevent blur event from hiding dropdown before selection triggers item.addEventListener('mousedown', (e) => { e.preventDefault(); searchInput.value = client.name; selectedClient = client; dropdownList.classList.add('hidden'); playProSound('select'); // Professional sound effect // Trigger any callback if needed if (typeof onClientSelected === 'function') onClientSelected(client); }); dropdownList.appendChild(item); }); } // Event Listeners for Search States searchInput.addEventListener('focus', () => { dropdownList.classList.remove('hidden'); renderList(clientsData); }); searchInput.addEventListener('input', (e) => { const query = e.target.value.toLowerCase().trim(); const filtered = clientsData.filter(c => c.name.toLowerCase().includes(query) || (c.phone && c.phone.includes(query)) ); dropdownList.classList.remove('hidden'); renderList(filtered); }); searchInput.addEventListener('blur', () => { // Small timeout to allow mousedown on dropdown items to fire first setTimeout(() => { dropdownList.classList.add('hidden'); }, 200); }); }
